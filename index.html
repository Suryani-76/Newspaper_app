Toggle regions. In-memory queries are cached independently. Cache-hits render immediately without loading skeletons.

// store/useRegionStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

export const useRegionStore = create()(
  persist((set) => ({
    followedCountries: [],
    activeCountry: 'GL',
    follow: (c) => set(s => ({ followedCountries: [...s.followedCountries, c] })),
    unfollow: (c) => set(s => ({ followedCountries: s.followedCountries.filter(x => x !== c) })),
    setActive: (c) => set({ activeCountry: c })
  }), { name: 'cinewire-regions' })
);
// hooks/useMovieNews.ts
import { useQuery } from '@tanstack/react-query';

export function useMovieNews(country) {
  return useQuery({
    queryKey: ['movieNews', country],
    queryFn: () => fetchNews(country),
    staleTime: 5 * 60 * 1000 // 5 minutes cache
  });
